# ATBClone CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a macOS native application cloning and sandbox isolation engine CLI tool in Python.

**Architecture:** A modular Python package with `cli`, `core`, `recipes`, and `executor` layers. It implements Soft Clone (symlink launcher) and Hard Clone (physical copy + PlistBuddy edit + wrapper hijack + resign) strategies driven by a YAML recipe engine, supporting per-clone proxy injection and auto-privilege escalation via `osascript`.

**Tech Stack:** Python 3.10+, Click (CLI), Pydantic (data validation), PyYAML (parsing), Rich (terminal UI).

## Global Constraints

- Target macOS native patterns, PySide6, Python 3.12+, with `conda run -n ATBClone` for Python env.
- Test command: `PYTHONPATH=src conda run -n ATBClone python -m pytest tests/`
- Default output directory is `~/Applications/` unless overridden.
- Hard clones into `/Applications` must use a single `osascript` admin prompt.
- Chromium-based apps (Chrome, Edge, Arc) MUST use soft cloning.
- Scripts generated by executor MUST include `set -e` for atomicity.

# src/atbclone/core/app_inspector.py
import subprocess
import re
from pathlib import Path
from typing import Tuple
from .models import AppInfo

class AppInspector:
    @staticmethod
    def _run_cmd(cmd: list[str]) -> str:
        try:
            return subprocess.check_output(cmd, stderr=subprocess.STDOUT, text=True).strip()
        except subprocess.CalledProcessError:
            return ""

    @classmethod
    def inspect(cls, app_path: str) -> AppInfo:
        path = Path(app_path)
        if not path.exists():
            raise FileNotFoundError(f"App not found: {app_path}")
        
        plist_path = path / "Contents" / "Info.plist"
        bundle_id = cls._run_cmd(["defaults", "read", str(plist_path), "CFBundleIdentifier"])
        app_name = cls._run_cmd(["defaults", "read", str(plist_path), "CFBundleName"])
        
        # Check sandbox
        entitlements = cls._run_cmd(["codesign", "-d", "--entitlements", "-", str(path)])
        has_sandbox = "com.apple.security.app-sandbox" in entitlements
        
        # Simple executable detection
        executable_name = cls._run_cmd(["defaults", "read", str(plist_path), "CFBundleExecutable"])
        executable = path / "Contents" / "MacOS" / executable_name
        
        return AppInfo(path=path, bundle_id=bundle_id, app_name=app_name, executable=executable, has_sandbox=has_sandbox)

    @staticmethod
    def next_available_name(app_name: str, dest_dir: Path) -> Tuple[str, int]:
        match = re.match(r"(.*?)(\d+)$", app_name)
        if match:
            base_name = match.group(1)
            num = int(match.group(2))
        else:
            base_name = app_name
            num = 1
        
        n = max(2, num) if num > 1 else 2
        while True:
            candidate = dest_dir / f"{base_name}{n}.app"
            if not candidate.exists():
                return f"{base_name}{n}", n
            n += 1
```

# src/atbclone/recipes/models.py
from pydantic import BaseModel, field_validator
from typing import Literal

class ProxyConfig(BaseModel):
    enabled: bool = False
    type: Literal["http", "socks5"] = "http"
    host: str = "127.0.0.1"
    port: int = 1080
    username: str = ""
    password: str = ""
    no_proxy: str = "localhost,127.0.0.1,*.local"

    @property
    def url(self) -> str:
        auth = f"{self.username}:{self.password}@" if self.username else ""
        return f"{self.type}://{auth}{self.host}:{self.port}"

class Recipe(BaseModel):
    bundle_id: str
    app_name: str
    strategy: Literal["hard_clone", "soft_clone"]
    strip_sandbox: bool = False
    proxy: ProxyConfig = ProxyConfig()
    environment_injection: dict[str, str] = {}
    symlink_whitelist: list[str] = []
    launch_args: list[str] = []

    @field_validator("strategy", mode="before")
    @classmethod
    def chromium_guard(cls, v, info):
        bid = info.data.get("bundle_id", "")
        if any(k in bid.lower() for k in ["chrome", "chromium", "microsoft.edge", "arc"]):
            return "soft_clone"
        return v
```

# src/atbclone/recipes/loader.py
import yaml
from pathlib import Path
from .models import Recipe

class RecipeLoader:
    BUILTIN_DIR = Path(__file__).parent / "builtin"
    
    @classmethod
    def match(cls, bundle_id: str) -> Recipe:
        # Priority: Local > Builtin
        local_dir = Path.home() / ".atbclone" / "recipes"
        local_file = local_dir / f"{bundle_id}.yaml"
        builtin_file = cls.BUILTIN_DIR / f"{bundle_id}.yaml"
        
        if local_file.exists():
            return cls._load_file(local_file)
        if builtin_file.exists():
            return cls._load_file(builtin_file)
            
        # Fallback default
        return Recipe(
            bundle_id=bundle_id,
            app_name="Unknown",
            strategy="hard_clone"
        )
        
    @staticmethod
    def _load_file(path: Path) -> Recipe:
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)
        return Recipe(**data)
```

# src/atbclone/executor/runner.py
import subprocess
import shlex

class CloneError(Exception):
    pass

class Runner:
    @staticmethod
    def run(script: str, needs_admin: bool) -> str:
        if needs_admin:
            return Runner._run_as_admin(script)
        else:
            return Runner._run_direct(script)
            
    @staticmethod
    def _run_direct(script: str) -> str:
        try:
            return subprocess.check_output(script, shell=True, stderr=subprocess.STDOUT, text=True)
        except subprocess.CalledProcessError as e:
            raise CloneError(f"Command failed (exit {e.returncode}):\n{e.output}")

    @staticmethod
    def _run_as_admin(script: str) -> str:
        escaped = script.replace('\\', '\\\\').replace('"', '\\"')
        applescript = f'do shell script "{escaped}" with administrator privileges'
        try:
            return subprocess.check_output(["/usr/bin/osascript", "-e", applescript], stderr=subprocess.STDOUT, text=True)
        except subprocess.CalledProcessError as e:
            raise CloneError(f"Admin command failed:\n{e.output}")
```

# src/atbclone/core/engines.py
import shlex
import textwrap
from .clone_task import CloneTask
from atbclone.executor.runner import Runner

class CloneEngine:
    @staticmethod
    def _build_proxy_env(task: CloneTask) -> str:
        proxy = task.recipe.proxy
        if not proxy.enabled:
            return ""
        return textwrap.dedent(f'''
            export HTTP_PROXY="{proxy.url}"
            export HTTPS_PROXY="{proxy.url}"
            export http_proxy="$HTTP_PROXY"
            export https_proxy="$HTTPS_PROXY"
            export NO_PROXY="{proxy.no_proxy}"
            export no_proxy="$NO_PROXY"
        ''').strip()

class SoftCloneEngine(CloneEngine):
    @classmethod
    def execute(cls, task: CloneTask, needs_admin: bool):
        src_bin = shlex.quote(str(task.source.executable))
        dst_mac = shlex.quote(str(task.dest_path / "Contents" / "MacOS"))
        dst_plist = shlex.quote(str(task.dest_path / "Contents" / "Info.plist"))
        wrapper = shlex.quote(str(task.dest_path / "Contents" / "MacOS" / task.source.app_name))
        
        args = " ".join([arg.replace("{{ATB_DATA_DIR}}", str(task.data_dir)) for arg in task.recipe.launch_args])
        
        script = f"""
set -e
mkdir -p {dst_mac}
cp {shlex.quote(str(task.source.path / "Contents" / "Info.plist"))} {dst_plist}
/usr/libexec/PlistBuddy -c "Set :CFBundleIdentifier {task.new_bundle_id}" {dst_plist}
/usr/libexec/PlistBuddy -c "Set :CFBundleName {task.clone_name}" {dst_plist}
cat << 'WRAPPER_EOF' > {wrapper}
#!/bin/bash
{cls._build_proxy_env(task)}
exec {src_bin} {args} >/dev/null 2>&1 &
WRAPPER_EOF
chmod +x {wrapper}
"""
        Runner.run(script, needs_admin)

class HardCloneEngine(CloneEngine):
    @classmethod
    def execute(cls, task: CloneTask, needs_admin: bool):
        src = shlex.quote(str(task.source.path))
        dst = shlex.quote(str(task.dest_path))
        dst_plist = shlex.quote(str(task.dest_path / "Contents" / "Info.plist"))
        orig_bin_name = task.source.executable.name
        bin_orig = shlex.quote(str(task.dest_path / "Contents" / "MacOS" / orig_bin_name))
        bin_bak = shlex.quote(str(task.dest_path / "Contents" / "MacOS" / f"{orig_bin_name}.bin"))
        wrapper = bin_orig
        
        env_vars = "\\n".join([f'export {k}="{v.replace("{{ATB_DATA_DIR}}", str(task.data_dir))}"' 
                              for k, v in task.recipe.environment_injection.items()])
        
        script = f"""
set -e
cp -R {src} {dst}
/usr/libexec/PlistBuddy -c "Set :CFBundleIdentifier {task.new_bundle_id}" {dst_plist}
/usr/libexec/PlistBuddy -c "Set :CFBundleName {task.clone_name}" {dst_plist}
mv {bin_orig} {bin_bak}
cat << 'WRAPPER_EOF' > {wrapper}
#!/bin/bash
{env_vars}
{cls._build_proxy_env(task)}
exec "$(dirname "\\$0")/{orig_bin_name}.bin" "\\$@"
WRAPPER_EOF
chmod +x {wrapper}
xattr -cr {dst}
"""
        if task.recipe.strip_sandbox:
            script += f"""
codesign -d --entitlements :- {dst} > /tmp/atb_entitlements.plist 2>/dev/null || true
/usr/libexec/PlistBuddy -c "Delete :com.apple.security.app-sandbox" /tmp/atb_entitlements.plist || true
codesign --force --deep --sign - --entitlements /tmp/atb_entitlements.plist {dst}
"""
        else:
            script += f"codesign --force --deep --sign - {dst}\\n"
            
        script += f"codesign -vv --deep --strict {dst}\\n"
        Runner.run(script, needs_admin)
```

# src/atbclone/cli/cmd_clone.py
import click
from pathlib import Path
from atbclone.core.app_inspector import AppInspector
from atbclone.recipes.loader import RecipeLoader
from atbclone.core.clone_task import CloneTask
from atbclone.core.engines import SoftCloneEngine, HardCloneEngine
from rich.console import Console

console = Console()

@click.command()
@click.argument("app_path", type=click.Path(exists=True))
@click.option("--name", help="分身名称")
@click.option("--output-dir", default=str(Path.home() / "Applications"), help="输出目录")
def clone(app_path, name, output_dir):
    """克隆应用"""
    out_path = Path(output_dir)
    out_path.mkdir(parents=True, exist_ok=True)
    
    info = AppInspector.inspect(app_path)
    recipe = RecipeLoader.match(info.bundle_id)
    
    clone_name, num = AppInspector.next_available_name(name or info.app_name, out_path)
    new_bundle_id = f"{info.bundle_id}.atb{num}"
    dest_path = out_path / f"{clone_name}.app"
    data_dir = Path.home() / ".atbclone" / "Data" / clone_name
    
    task = CloneTask(info, dest_path, data_dir, recipe, clone_name, new_bundle_id)
    needs_admin = not dest_path.is_relative_to(Path.home())
    
    console.print(f"[bold green]Starting clone:[/bold green] {info.app_name} -> {clone_name}")
    try:
        if recipe.strategy == "soft_clone":
            SoftCloneEngine.execute(task, needs_admin)
        else:
            HardCloneEngine.execute(task, needs_admin)
        console.print(f"[bold green]Success![/bold green] Clone created at {dest_path}")
    except Exception as e:
        console.print(f"[bold red]Error:[/bold red] {e}")
```
